cmake_minimum_required(VERSION 3.18)
project(launch_native_examples LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(LICENTIVO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LICENTIVO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../sdk/native launch-account)
add_executable(launch_c ../c/main.c)
target_link_libraries(launch_c PRIVATE launch_account)
add_executable(launch_cpp ../cpp/main.cpp)
target_link_libraries(launch_cpp PRIVATE LicentivoLaunch::CPP)
