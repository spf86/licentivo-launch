cmake_minimum_required(VERSION 3.18)
project(licentivo_launch VERSION 0.1.6 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(LICENTIVO_LAUNCH_CURL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/curl")
if(WIN32 AND EXISTS "${LICENTIVO_LAUNCH_CURL_ROOT}/include/curl/curl.h")
  set(CURL_ROOT "${LICENTIVO_LAUNCH_CURL_ROOT}")
endif()
if(NOT TARGET Licentivo::C)
  set(LICENTIVO_BUILD_EXAMPLES OFF)
  set(LICENTIVO_BUILD_TESTS OFF)
  add_subdirectory(vendor/sdk launch-sdk)
endif()
add_library(launch_account STATIC account.c)
add_library(LicentivoLaunch::Account ALIAS launch_account)
set_target_properties(launch_account PROPERTIES EXPORT_NAME Account C_STANDARD 11)
target_include_directories(launch_account PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/licentivo-launch>)
target_link_libraries(launch_account PUBLIC Licentivo::C)
if(WIN32)
  target_link_libraries(launch_account PUBLIC crypt32)
endif()
add_library(launch_cpp_interface INTERFACE)
add_library(LicentivoLaunch::CPP ALIAS launch_cpp_interface)
set_target_properties(launch_cpp_interface PROPERTIES EXPORT_NAME CPP)
target_link_libraries(launch_cpp_interface INTERFACE launch_account)
target_compile_features(launch_cpp_interface INTERFACE cxx_std_17)
option(LICENTIVO_LAUNCH_BUILD_TESTS "Build Launch lifecycle regression drivers" OFF)
if(LICENTIVO_LAUNCH_BUILD_TESTS)
  add_executable(launch_native_test tests/application_test.c)
  target_link_libraries(launch_native_test PRIVATE Licentivo::C)
  if(WIN32)
    target_link_libraries(launch_native_test PRIVATE crypt32)
  endif()
  add_executable(launch_cpp_test tests/application_test.cpp)
  target_compile_features(launch_cpp_test PRIVATE cxx_std_17)
  target_link_libraries(launch_cpp_test PRIVATE LicentivoLaunch::CPP)
endif()
install(TARGETS launch_account launch_cpp_interface EXPORT LicentivoLaunchTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES account.h launch.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/licentivo-launch)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/licentivo-launch)
if(WIN32 AND EXISTS "${LICENTIVO_LAUNCH_CURL_ROOT}/include/curl/curl.h")
  install(DIRECTORY "${LICENTIVO_LAUNCH_CURL_ROOT}/" DESTINATION third_party/curl)
endif()
include(LaunchRuntime.cmake)
install(FILES LaunchRuntime.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LicentivoLaunch)
install(EXPORT LicentivoLaunchTargets NAMESPACE LicentivoLaunch:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LicentivoLaunch)
configure_package_config_file(LicentivoLaunchConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/LicentivoLaunchConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LicentivoLaunch)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/LicentivoLaunchConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/LicentivoLaunchConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/LicentivoLaunchConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/LicentivoLaunch)
